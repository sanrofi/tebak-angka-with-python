Game Tebak Angka

Saya telah menyimpan angka dari 1 sampai 10. Coba tebak!

import random from pyscript import document angka_rahasia = random.randint(1, 10) def cek_tebakan(event): input_elemen = document.querySelector("#tebakan") hasil_elemen = document.querySelector("#hasil") tebakan = int(input_elemen.value) if tebakan == angka_rahasia: hasil_elemen.innerText = "Benar! Anda menang!" elif tebakan < angka_rahasia: hasil_elemen.innerText = "Terlalu kecil, coba lagi!" else: hasil_elemen.innerText = "Terlalu besar, coba lagi!"
